function validateAnnotation(a){
 if(!a||typeof a!=='object'||Array.isArray(a))throw Error('Invalid annotation record.');
 if(typeof a.id!=='string'||!/^[A-Za-z0-9_-]{1,100}$/.test(a.id))throw Error('Invalid annotation ID.');
 if(!finite(a.start_s)||!finite(a.end_s)||a.start_s<0||a.end_s<a.start_s||a.end_s>duration)throw Error('Use 0 <= start <= end <= session duration; equal times define a point.');
 const result={id:a.id,start_s:a.start_s,end_s:a.end_s};
 for(const key of Object.keys(annotationFields)){
  if(typeof a[key]!=='string'||a[key].length>(['quote','evidence','notes'].includes(key)?10000:120))throw Error('Invalid or oversized annotation text: '+key);
  result[key]=a[key];
 }
 if(!a.label.trim()||!a.annotator_id.trim())throw Error('Label and annotator ID are required.');
 if(!annotationSources.includes(a.source)||!annotationConfidences.includes(a.confidence))throw Error('Unknown evidence source or confidence.');
 result.review_status=a.review_status??'draft';
 if(!['draft','reviewed','disputed','excluded'].includes(result.review_status))throw Error('Invalid review status.');
 result.participant_intensity=a.participant_intensity??null;

 result.intensity_source=result.participant_intensity===null?null:'participant_self_report';
 result.confusion_rating_1_5=a.confusion_rating_1_5??null;
 if(result.confusion_rating_1_5!==null&&(!Number.isInteger(result.confusion_rating_1_5)||result.confusion_rating_1_5<1||result.confusion_rating_1_5>5))throw Error('Confusion rating must be blank or a whole number from 1 to 5.');
 result.confusion_rating_source=result.confusion_rating_1_5===null?null:'participant_retrospective_self_report';
 result.related_observation_id=a.related_observation_id??'';
 if(typeof result.related_observation_id!=='string'||(result.related_observation_id&&!/^obs-[a-f0-9]{32}$/.test(result.related_observation_id)))throw Error('Invalid linked observation ID.');
 result.related_ids=a.related_ids??[];
 if(!Array.isArray(result.related_ids)||result.related_ids.length>50||result.related_ids.some(id=>typeof id!=='string'||!/^[A-Za-z0-9_-]{1,100}$/.test(id)))throw Error('Invalid related annotation IDs.');
 result.record_revision=a.record_revision??1;
 if(!Number.isSafeInteger(result.record_revision)||result.record_revision<1)throw Error('Invalid record revision.');
 result.revisions=a.revisions??[];
 if(!Array.isArray(result.revisions)||result.revisions.length>100)throw Error('Maximum 100 historical revisions per record.');

 if(Date.parse(a.updated_utc)<Date.parse(a.created_utc))throw Error('Annotation update precedes creation.');
 result.video_time_s=null;result.video_offset_s=null;

 }
 return result;
}

function targetMetrics(list,a,b){
 const result=new Map();let valid=0,covered=0,previous=null;
 for(const s of list){
  if(s.t<a){previous=null;continue;}if(s.t>=b)break;
  if(s.valid)valid++;
  if(s.id&&s.status==='target_hit'&&s.valid){
   if(!result.has(s.id))result.set(s.id,{id:s.id,first:s.t,count:0,visits:0,seconds:0});
   const m=result.get(s.id);m.count++;
   if(!previous||!previous.valid||previous.id!==s.id||previous.status!=='target_hit'||previous.source!==s.source||previous.calibration!==s.calibration||s.t-previous.t>.1)m.visits++;
  }
  const dt=previous?s.t-previous.t:0;
  if(previous&&previous.valid&&s.valid&&dt>0&&dt<=.1&&previous.source===s.source&&previous.calibration===s.calibration){
   covered+=dt;
   // Attribute time only when consecutive callbacks agree on the same target.
   if(s.id&&previous.id===s.id&&previous.status==='target_hit'&&s.status==='target_hit')result.get(s.id).seconds+=dt;
  }
  previous=s;
 }
 return {targets:[...result.values()],valid,covered};
}
